cmake_minimum_required(VERSION 3.22.1)
project(lanmonitor)
add_library(monitor SHARED receiver.cpp forward.cpp)
set(NDI_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../ndi-sdk")
set(NDI_SO "${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI}/libndi.so")
if(EXISTS "${NDI_ROOT}/include/Processing.NDI.Lib.h" AND EXISTS "${NDI_SO}")
add_library(ndi SHARED IMPORTED)
set_target_properties(ndi PROPERTIES IMPORTED_LOCATION "${NDI_SO}")
target_include_directories(monitor PRIVATE "${NDI_ROOT}/include")
target_compile_definitions(monitor PRIVATE HAS_NDI=1)
target_link_libraries(monitor ndi)
else()
message(FATAL_ERROR "withNdi requires the official Android SDK headers and runtime; see README")
endif()
target_link_libraries(monitor android log)
